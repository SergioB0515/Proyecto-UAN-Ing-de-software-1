import hmac, hashlib, time

def valido(cuerpo_crudo, cabecera, secreto, tolerancia=300):
    partes = dict(p.split("=", 1) for p in cabecera.split(","))
    base = f"{partes['t']}.{cuerpo_crudo}".encode()
    esperada = hmac.new(secreto.encode(), base, hashlib.sha256).hexdigest()
    reciente = abs(time.time() - int(partes["t"])) < tolerancia
    return reciente and hmac.compare_digest(esperada, partes["v1"])
