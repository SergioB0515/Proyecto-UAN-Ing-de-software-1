{{ _('Ejemplos') }}

# {{ _('Listar mis tickets abiertos') }}
curl -H "Authorization: Bearer $TOKEN" \
     "{{ base }}/api/v1/tickets?estado=abierto"

# {{ _('Crear un ticket') }}
curl -X POST -H "Authorization: Bearer $TOKEN" \
     -H "Content-Type: application/json" \
     -d '{"texto": "La impresora del piso 3 no imprime desde esta mañana"}' \
     "{{ base }}/api/v1/tickets"
import requests

API = "{{ base }}/api/v1"
cabeceras = {"Authorization": "Bearer " + TOKEN}

r = requests.get(f"{API}/tickets", params={"estado": "abierto"}, headers=cabeceras)
for t in r.json()["datos"]:
    print(t["id"], t["prioridad"], t["texto"][:60])

nuevo = requests.post(f"{API}/tickets", json={"texto": "No puedo acceder a la carpeta compartida de finanzas"}, headers=cabeceras)
print(nuevo.status_code, nuevo.json()["categoria"])
const API = "{{ base }}/api/v1";
const cabeceras = { Authorization: `Bearer ${TOKEN}`, "Content-Type": "application/json" };

const r = await fetch(`${API}/tickets?estado=abierto`, { headers: cabeceras });
const { datos, paginacion } = await r.json();
console.log(paginacion.total, datos.map(t => t.id));
